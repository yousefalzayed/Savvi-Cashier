async function loadOffers() {

    sb.from('categories').select('id, slug, name_en, name_ar, sort_order').order('sort_order'),
    sb.from('merchants').select('id, name_en, name_ar, category_id'),
    sb.from('branches').select('merchant_id, name, area'),
    sb.from('offers').select('id, merchant_id, category_id, title_en, title_ar, min_bill_amount, starts_at, ends_at')
  ]);
