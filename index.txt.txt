  const s = new Html5Qrcode('reader', { verbose: false });
  try {
    await s.start({ facingMode: 'environment' }, { fps: 10, qrbox: { width: 230, height: 230 } }, (text) => gotToken(text), () => {});
    scanner = s;
    if (S.view !== 'scan' || S.manual) stopScanner();
  } catch (e) {
    S.manual = true; S.err = t('cameraBlocked'); if (S.view === 'scan') render();
  } finally { starting = false; }
}
function stopScanner() {
  const s = scanner; scanner = null;
  if (!s) return Promise.resolve();
  return s.stop().then(() => { try { s.clear(); } catch (e) {} }).catch(() => {});
}

function render() {
  document.documentElement.lang = S.lang; document.documentElement.dir = S.lang === 'ar' ? 'rtl' : 'ltr';
  const keepBill = S.view === 'bill' ? document.activeElement && document.activeElement.id === 'bill' : false;
  const restart = !!scanner;
  stopScanner();
  document.getElementById('app').innerHTML = views[S.view]();
  if (S.view === 'scan' && !S.manual) setTimeout(startScanner, restart ? 400 : 50);
  if (S.view === 'code') {
    const boxes = Array.from(document.querySelectorAll('.codes input'));
    boxes.forEach((b, i) => {
      b.addEventListener('input', () => {
        const v = b.value.replace(/\D/g, '');
        if (v.length > 1) { v.slice(0, 6).split('').forEach((d, j) => { if (boxes[j]) boxes[j].value = d; }); (boxes[Math.min(v.length, 6) - 1]).focus(); return; }
        b.value = v; if (v && boxes[i + 1]) boxes[i + 1].focus();
      });
      b.addEventListener('keydown', (e) => { if (e.key === 'Backspace' && !b.value && boxes[i - 1]) boxes[i - 1].focus(); });
    });
    boxes[0] && boxes[0].focus();
  }
  if (keepBill) { const el = document.getElementById('bill'); el.focus(); el.setSelectionRange(el.value.length, el.value.length); }
}
